```java
                                @Param("limit") int limit);

    List<RankRow> violationRank(@Param("from") LocalDate from,
                                @Param("to") LocalDate to,
                                @Param("limit") int limit);

    List<RankRow> hourlyHeat(@Param("from") LocalDate from,
                             @Param("to") LocalDate to);

```

`src/main/resources/mapper/ReportMapper.xml`：

```xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE mapper PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN"
        "https://mybatis.org/dtd/mybatis-3-mapper.dtd">
<mapper namespace="com.seat.reserve.mapper.ReportMapper">

    <select id="overview" resultType="com.seat.reserve.web.dto.OverviewReport">
        SELECT
            COUNT(*) AS total,
            SUM(CASE WHEN status = 'FINISHED'  THEN 1 ELSE 0 END) AS finished,
            SUM(CASE WHEN status = 'EXPIRED'   THEN 1 ELSE 0 END) AS expired,
            SUM(CASE WHEN status = 'CANCELLED' THEN 1 ELSE 0 END) AS cancelled,
            ROUND(
                SUM(CASE WHEN status = 'FINISHED' THEN 1 ELSE 0 END) /
                NULLIF(SUM(CASE WHEN status IN ('FINISHED','EXPIRED') THEN 1 ELSE 0 END), 0)
            , 3) AS fulfillRate,
            ROUND(SUM(TIME_TO_SEC(TIMEDIFF(end_time, start_time))) / 3600, 1) AS bookedHours
        FROM seat_booking
        WHERE book_date BETWEEN #{from} AND #{to}
    </select>

    <select id="seatUsageRank" resultType="com.seat.reserve.web.dto.RankRow">
        SELECT b.seat_id AS id, s.seat_no AS name, r.name AS extra,
               COUNT(*) AS times,
               0 AS noShow,
               ROUND(SUM(TIME_TO_SEC(TIMEDIFF(b.end_time, b.start_time))) / 3600, 1) AS hours
        FROM seat_booking b
        JOIN lib_seat s ON s.id = b.seat_id
        JOIN lib_room r ON r.id = b.room_id
        WHERE b.book_date BETWEEN #{from} AND #{to}
          AND b.status IN ('BOOKED','CHECKED_IN','LEAVING','FINISHED')
        GROUP BY b.seat_id, s.seat_no, r.name
        ORDER BY hours DESC, times DESC
        LIMIT #{limit}
    </select>

    <select id="violationRank" resultType="com.seat.reserve.web.dto.RankRow">
        SELECT v.user_id AS id, u.nickname AS name, u.username AS extra,
               COUNT(*) AS times,
               SUM(CASE WHEN v.type = 'NO_SHOW'  THEN 1 ELSE 0 END) AS noShow,
               0 AS hours
        FROM violation v
        JOIN sys_user u ON u.id = v.user_id
        WHERE v.created_at &gt;= #{from}
          AND v.created_at &lt;  DATE_ADD(#{to}, INTERVAL 1 DAY)
        GROUP BY v.user_id, u.nickname, u.username
        ORDER BY times DESC
        LIMIT #{limit}
    </select>

    <select id="hourlyHeat" resultType="com.seat.reserve.web.dto.RankRow">
        SELECT HOUR(start_time) AS id,
               CONCAT(HOUR(start_time), ':00') AS name,
               '' AS extra,
               COUNT(*) AS times,
               0 AS noShow,
               0 AS hours
        FROM seat_booking
        WHERE book_date BETWEEN #{from} AND #{to}
          AND status IN ('BOOKED','CHECKED_IN','LEAVING','FINISHED')
        GROUP BY HOUR(start_time)
        ORDER BY id
    </select>

</mapper>
```

**指标口径（面试会问，先想清楚）**

- 履约率分母**排除 CANCELLED**——用户主动取消不是平台失约。用 `NULLIF` 防止除零。
- 座位使用率这里用"累计预约小时数"绝对值表示，不强行造分母。要算百分比的话，分母 = 座位数 × 天数 × (close_time − open_time)，需要 JOIN `lib_room` 并用 `TIMESTAMPDIFF`，但对种子数据量意义不大，先不做。
- `hourlyHeat` 复用 `RankRow`，`id` 字段装小时数——有点取巧但省一个类，简历里不必提。

### 1.5 第 4 步：ReportService

`service/ReportService.java`：

```java
package com.seat.reserve.service;

import com.seat.reserve.common.BizException;
import com.seat.reserve.mapper.ReportMapper;
import com.seat.reserve.web.dto.OverviewReport;
import com.seat.reserve.web.dto.RankRow;
import org.springframework.stereotype.Service;

import java.time.LocalDate;
import java.util.List;

@Service
public class ReportService {

    private static final int MAX_RANGE_DAYS = 92;   // 防止一次扫全表

    private final ReportMapper reportMapper;

    public ReportService(ReportMapper reportMapper) {
        this.reportMapper = reportMapper;
    }

    public OverviewReport overview(LocalDate from, LocalDate to) {
        checkRange(from, to);
        return reportMapper.overview(from, to);
    }

    public List<RankRow> seatUsage(LocalDate from, LocalDate to) {
        checkRange(from, to);
        return reportMapper.seatUsageRank(from, to, 20);
    }

    public List<RankRow> violationRank(LocalDate from, LocalDate to) {
        checkRange(from, to);
        return reportMapper.violationRank(from, to, 20);
    }

    public List<RankRow> hourlyHeat(LocalDate from, LocalDate to) {
        checkRange(from, to);
        return reportMapper.hourlyHeat(from, to);
    }

    private void checkRange(LocalDate from, LocalDate to) {
        if (from == null || to == null || from.isAfter(to)) {
            throw BizException.badRequest("日期区间不合法");
        }
        if (from.plusDays(MAX_RANGE_DAYS).isBefore(to)) {
            throw BizException.badRequest("单次最多查询 " + MAX_RANGE_DAYS + " 天");
        }
    }
}
```

## 2. 阶段二：通知触达（事务发件箱）

### 2.1 数据流

```
业务方法（事务内）
  ├─ 改状态（CAS UPDATE）
  ├─ 写 booking_log
  └─ 写 notification_task   ← 同一事务，要么全成功要么全回滚
                    ↓
NotificationWorker（@Scheduled，每 30s）
  ├─ Redis 锁（多实例互斥，复用 RedisLock）
  ├─ 扫 PENDING 且 next_retry_at <= now
  ├─ 查内容 → 发送（LOG 或 EMAIL 策略）
  ├─ 成功 → status=SENT, sent_at=NOW()
  └─ 失败 → retry_count+1，指数退避；超 3 次 → FAILED
```

**投递语义**：at-least-once（至少一次）。靠唯一键 `(booking_id, event_type)` 防重复建任务，靠 `status` 状态机防重复发送。这个术语面试要能主动说出来。

### 2.3 第 1 步：发件箱 Mapper

`mapper/NotificationTaskMapper.java`：

```java
package com.seat.reserve.mapper;

import com.seat.reserve.domain.NotificationTask;
import org.apache.ibatis.annotations.Param;

import java.util.List;

public interface NotificationTaskMapper {

    /** 业务事务内调用：INSERT IGNORE 靠唯一键实现幂等 */
    int insertIgnore(NotificationTask task);

    /** worker 扫描待投递任务 */
    List<NotificationTask> pollPending(@Param("limit") int limit);

    /** 投递成功 */
    int markSent(@Param("id") Long id);

    /** 投递失败：次数 +1，计算下次重试时间 */
    int markFailed(@Param("id") Long id,
                   @Param("nextRetryAt") java.time.LocalDateTime nextRetryAt,
                   @Param("giveUp") boolean giveUp);
}
```

```xml
<!DOCTYPE mapper PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN"
        "https://mybatis.org/dtd/mybatis-3-mapper.dtd">
<mapper namespace="com.seat.reserve.mapper.NotificationTaskMapper">

    <insert id="insertIgnore">
        INSERT IGNORE INTO notification_task
        (booking_id, user_id, event_type, channel, status, next_retry_at)
        VALUES (#{bookingId}, #{userId}, #{eventType}, #{channel}, 'PENDING', NOW())
    </insert>

    <select id="pollPending" resultType="com.seat.reserve.domain.NotificationTask">
        SELECT id, booking_id, user_id, event_type, channel, status, retry_count, next_retry_at
        FROM notification_task
        WHERE status = 'PENDING' AND next_retry_at &lt;= NOW()
        ORDER BY next_retry_at
        LIMIT #{limit}
    </select>

    <update id="markSent">
        UPDATE notification_task
        SET status = 'SENT', sent_at = NOW()
        WHERE id = #{id} AND status = 'PENDING'
    </update>

    <update id="markFailed">
        UPDATE notification_task
        SET status      = IF(#{giveUp}, 'FAILED', 'PENDING'),
            retry_count = retry_count + 1,
            next_retry_at = #{nextRetryAt}
        WHERE id = #{id} AND status = 'PENDING'
    </update>

</mapper>
```

### 2.4 第 2 步：入队服务（在业务事务内调用）

```java
package com.seat.reserve.service;

import com.seat.reserve.domain.NotificationTask;
import com.seat.reserve.mapper.NotificationTaskMapper;
import org.springframework.stereotype.Service;

@Service
public class NotificationService {

    public static final int MAX_RETRY = 3;

    private final NotificationTaskMapper mapper;

    public NotificationService(NotificationTaskMapper mapper) {
        this.mapper = mapper;
    }

    /**
     * 必须在业务事务内调用：发件箱记录与状态变更同生共死。
     * 靠 INSERT IGNORE + 唯一键实现幂等，重复入队不会报错也不会产生第二条。

    public void enqueue(Long bookingId, Long userId, String eventType) {
        NotificationTask task = new NotificationTask();
        task.setBookingId(bookingId);
        task.setUserId(userId);
        task.setEventType(eventType);
        task.setChannel("LOG");   // 阶段二先用日志通道，邮件通道见 2.7
        mapper.insertIgnore(task);
    }
}
```

`service/NotificationWorker.java`（照搬 `DelayScheduler` 的模式）：

```java
package com.seat.reserve.service;

import com.seat.reserve.common.RedisLock;
import com.seat.reserve.domain.NotificationTask;
import com.seat.reserve.mapper.NotificationTaskMapper;
import com.seat.reserve.mapper.BookingMapper;
import lombok.extern.slf4j.Slf4j;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;

import java.time.Duration;
import java.time.LocalDateTime;
import java.util.List;
import java.util.Map;

@Slf4j
@Component
public class NotificationWorker {

    private static final int BATCH = 100;
    private static final Duration LOCK_TTL = Duration.ofSeconds(25);
    private static final Map<String, String> TEMPLATES = Map.of(
            "BOOKED",      "预约成功，请按时签到",
            "CANCELLED",   "预约已取消",
            "CHECKED_IN",  "签到成功，祝你学习愉快",
            "NO_SHOW",     "超时未签到，已记违约",
            "OVERTIME",    "暂离超时，已记违约",
            "FINISHED",    "预约已完成"
    );

    private final NotificationTaskMapper taskMapper;
    private final BookingMapper bookingMapper;
    private final RedisLock redisLock;
    private final List<NotificationSender> senders;

    public NotificationWorker(NotificationTaskMapper taskMapper, BookingMapper bookingMapper,
                              RedisLock redisLock, List<NotificationSender> senders) {
        this.taskMapper = taskMapper;
        this.bookingMapper = bookingMapper;
        this.redisLock = redisLock;
        this.senders = senders;
    }

    @Scheduled(fixedDelay = 30_000, initialDelay = 10_000)
    public void dispatch() {
        String lockKey = "lock:notify:dispatch";
        String token = redisLock.tryLock(lockKey, LOCK_TTL);
        if (token == null) return;
        try {
            List<NotificationTask> tasks = taskMapper.pollPending(BATCH);
            for (NotificationTask task : tasks) {
                // 状态变更可能还没提交就扫到了？不会：写入在业务事务里，提交后才可见（READ_COMMITTED）
                String content = TEMPLATES.getOrDefault(task.getEventType(), "预约状态更新");
                try {
                    senders.stream()
                           .filter(s -> s.channel().equals(task.getChannel()))
                           .forEach(s -> s.send(task, content));
                    taskMapper.markSent(task.getId());
                } catch (Exception e) {
                    int retries = task.getRetryCount() == null ? 0 : task.getRetryCount();
                    boolean giveUp = retries + 1 >= NotificationService.MAX_RETRY;
                    long backoffSeconds = 1L << Math.min(retries, 5);   // 1,2,4,8,16,32 秒
                    taskMapper.markFailed(task.getId(),
                            LocalDateTime.now().plusSeconds(backoffSeconds), giveUp);
                    log.error("通知 {} 投递失败（第 {} 次），{}",
                            task.getId(), retries + 1, giveUp ? "放弃" : "稍后重试", e);
                }
            }
        } catch (Exception e) {
            log.error("通知扫描失败", e);
        } finally {
            redisLock.unlock(lockKey, token);
        }
    }
}
```

**这里三个面试点**：
1. **指数退避** `1L << retries`——失败越多，重试间隔越长，避免雪崩。
2. **多实例互斥**——`@Scheduled` 每个实例都会跑，靠 Redis 锁保证只有一个在投递。
3. **失败不阻塞**——单条失败被 catch 住，同批其他任务继续；`markSent` 带 `status='PENDING'` 条件，天然幂等。

## 6. 常见坑

| 现象 | 原因 | 解决 |
|---|---|---|
| 访问 `/admin` 404 | `SpaForwardController` 没加 `/admin` | 按 1.7 改 |
| 访问 `/admin` 401 | `SecurityConfig` 白名单没加 `/admin` | 按 1.7 改 |
| 报表接口 500，说列名不存在 | MyBatis 驼峰映射没覆盖别名 | `application.yaml` 已有 `map-underscore-to-camel-case: true`，确认别名写成 `fulfillRate` 这种驼峰 |
| Flyway 启动报 checksum mismatch | 改了已执行的 V1/V2/V3 | 已执行的迁移脚本绝不改动，需要变更只能新增 V4 |
| 履约率是 0.000 | 种子数据只有 BOOKED，没有 FINISHED | 手动把那条预约签到、再等延迟队列置 FINISHED，或补几条种子数据 |

---

**改动总原则**：预约内核（三层防重 + CAS 状态机 + ZSet 延迟队列）一个字都不改。报表以"旁路"方式读状态流水，不侵入主链路。这样面试时被追问底层，你的答案依然是那套你最熟的东西。
